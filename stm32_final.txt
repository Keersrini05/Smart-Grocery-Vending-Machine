#include <stdio.h>
#include <string.h>

/* ================= RCC ================= */
#define RCC_BASE        0x40023800UL
#define RCC_AHB1ENR     (*(volatile unsigned long *)(RCC_BASE + 0x30))
#define RCC_APB2ENR     (*(volatile unsigned long *)(RCC_BASE + 0x44))

/* ================= GPIO ================= */
#define GPIOA_BASE 0x40020000UL
#define GPIOA_MODER (*(volatile unsigned long *)(GPIOA_BASE+0x00))
#define GPIOA_IDR   (*(volatile unsigned long *)(GPIOA_BASE+0x10))
#define GPIOA_PUPDR (*(volatile unsigned long *)(GPIOA_BASE+0x0C))
#define GPIOA_AFRH  (*(volatile unsigned long *)(GPIOA_BASE+0x24))

#define GPIOB_BASE 0x40020400UL
#define GPIOB_MODER (*(volatile unsigned long *)(GPIOB_BASE+0x00))
#define GPIOB_ODR   (*(volatile unsigned long *)(GPIOB_BASE+0x14))
#define GPIOB_IDR   (*(volatile unsigned long *)(GPIOB_BASE+0x10))
#define GPIOB_BSRR  (*(volatile unsigned long *)(GPIOB_BASE+0x18))

#define GPIOC_BASE 0x40020800UL
#define GPIOC_MODER (*(volatile unsigned long *)(GPIOC_BASE+0x00))
#define GPIOC_ODR   (*(volatile unsigned long *)(GPIOC_BASE+0x14))
#define GPIOC_IDR   (*(volatile unsigned long *)(GPIOC_BASE+0x10))
#define GPIOC_PUPDR (*(volatile unsigned long *)(GPIOC_BASE+0x0C))

/* ================= USART1 ================= */
#define USART1_BASE 0x40011000UL
#define USART1_SR   (*(volatile unsigned long *)(USART1_BASE+0x00))
#define USART1_DR   (*(volatile unsigned long *)(USART1_BASE+0x04))
#define USART1_BRR  (*(volatile unsigned long *)(USART1_BASE+0x08))
#define USART1_CR1  (*(volatile unsigned long *)(USART1_BASE+0x0C))



// ========================
// LCD Pin Masks (GPIOB)
// ========================
#define RS   (1U << 9)   // PB9
#define EN   (1U << 10)  // PB10
#define D4   (1U << 12)  // PB12
#define D5   (1U << 13)  // PB13
#define D6   (1U << 14)  // PB14
#define D7   (1U << 15)  // PB15



// ========================
// Motor/LED Pin Masks (GPIOB)
// ========================
#define MOTOR_RICE   (1U << 4)   // PB4
#define MOTOR_DHALL  (1U << 5)   // PB5
#define MOTOR_WHEAT  (1U << 6)   // PB6



// ========================
// IR Sensor Pin Masks (GPIOA)
// ========================
#define IR_RICE   (1U << 0)  // PA0
#define IR_DALL  (1U << 1)  // PA1
#define IR_SUGAR  (1U << 4)  // PA2



// ========================
// Calibration
// ========================
long offset = 84585971;
long scale  = 88602456;
long known_weight = 1000;
long tare_offset = 0;
char selected_item = 0;
char selected_grams = 0;



// ========================
// Delay
// ========================
void delay(volatile unsigned int t) {
    for (; t > 0; t--);
}



// ========================
// Keypad
// ========================
char keypad_map[4][3] = {
    {'1','2','3'},
    {'4','5','6'},
    {'7','8','9'},
    {'*','0','#'}
};

char keypad_getkey(void)
{
    char key = 0;
    for (int row=0; row<4; row++) {
        GPIOC_ODR |= (0xF << 1);
        GPIOC_ODR &= ~(1U << (row+1));
        delay(1000);

    for (int col=0; col<3; col++) {
         if (!(GPIOC_IDR & (1U << (col+5))))
            {
             delay(50000);
             while (!(GPIOC_IDR & (1U << (col+5))));
             key = keypad_map[row][col];
             GPIOC_ODR |= (1U << (row+1));
             return key;
            }
        }
    }
    return 0;
}


// ========================
// LCD Functions
// ========================
void lcd_pulse_enable(void) {
    GPIOB_ODR |= EN;
    delay(2000);
    GPIOB_ODR &= ~EN;
    delay(2000);
}

void lcd_send_nibble(unsigned char data) {
    if (data & 0x01) GPIOB_ODR |= D4; else GPIOB_ODR &= ~D4;
    if (data & 0x02) GPIOB_ODR |= D5; else GPIOB_ODR &= ~D5;
    if (data & 0x04) GPIOB_ODR |= D6; else GPIOB_ODR &= ~D6;
    if (data & 0x08) GPIOB_ODR |= D7; else GPIOB_ODR &= ~D7;
    lcd_pulse_enable();
}

void lcd_send_byte(unsigned char data, int isData) {
    if (isData) GPIOB_ODR |= RS;
    else GPIOB_ODR &= ~RS;

    lcd_send_nibble(data >> 4);
    lcd_send_nibble(data & 0x0F);

    delay(5000);
}

void lcd_cmd(unsigned char cmd) { lcd_send_byte(cmd, 0); }
void lcd_data(unsigned char data) { lcd_send_byte(data, 1); }

void lcd_init(void) {
    delay(50000);
    lcd_send_nibble(0x03); delay(50000);
    lcd_send_nibble(0x03); delay(50000);
    lcd_send_nibble(0x03); delay(50000);
    lcd_send_nibble(0x02);

    lcd_cmd(0x28);
    lcd_cmd(0x0C);
    lcd_cmd(0x06);
    lcd_cmd(0x01);
    delay(50000);
}

void lcd_print(char *s) {
    while (*s) lcd_data(*s++);
}


char lcd_scroll_left_with_key(char *text) {
    int len=0; while(text[len]) len++;
    int i=0; char key=0;

    while(1) {
        char buf[17];
        for(int j=0;j<16;j++) buf[j]=text[(i+j)%len];
        buf[16]='\0';
        lcd_cmd(0xC0); lcd_print(buf);

        key = keypad_getkey();
        if(key=='1'||key=='2'||key=='3') return key;

        i++; if(i>=len) i=0;
        delay(600000);
    }
}



// ========================
// HX711
// ========================
unsigned long HX711_Read(void) {
    unsigned long count=0;
    while(GPIOB_IDR & (1<<0));

    for(int i=0;i<24;i++){
        GPIOB_BSRR = (1<<1);
        for(volatile int d=0;d<50;d++);
        count = count<<1;
        if(GPIOB_IDR & (1<<0)) count++;
        GPIOB_BSRR = (1<<(1+16));
        for(volatile int d=0;d<50;d++);
    }

    GPIOB_BSRR = (1<<1); for(volatile int d=0;d<50;d++);
    GPIOB_BSRR = (1<<(1+16));

    return count ^ 0x800000;
}

void tare(void) {
    long sum=0;
    for(int i=0;i<20;i++) sum += HX711_Read();
    tare_offset = sum/20;
}

int get_weight(long raw) {
    long diff = scale-offset; if(diff==0) return 0;
    long value = (raw-tare_offset)*known_weight/diff;
    if(value<0) value=0;
    return (int)value;
}



// ========================
// Motor Control
// ========================
void motor_control(int item)
{
    GPIOB_ODR |= (MOTOR_RICE|MOTOR_DHALL|MOTOR_WHEAT);

    if(item==1) GPIOB_ODR &= ~MOTOR_RICE;
    else if(item==2) GPIOB_ODR &= ~MOTOR_DHALL;
    else if(item==3) GPIOB_ODR &= ~MOTOR_WHEAT;
}

void back(void)
{
    char key=0;
    while(1){
        key = keypad_getkey();
        if(key=='*') break;
    }

    lcd_cmd(0x01);
}

// ========================
// Wait for '*' to continue
// ========================
void wait_for_continue(int lastWeight)
{
    GPIOB_ODR |= (MOTOR_RICE | MOTOR_DHALL | MOTOR_WHEAT);

    lcd_cmd(0x80);
    lcd_print("Press * to cont");

    lcd_cmd(0xC0);
    char buf[16];
    sprintf(buf,"grams: %03d",lastWeight);
    lcd_print(buf);
    back();

}



// ========================
// Uart communication
// ========================
void uart1_tx_char(char c)
{
    while(!(USART1_SR & (1<<7))); // TXE
    USART1_DR = c;
}


void uart1_tx_string(char *s)
{
    while(*s)
        uart1_tx_char(*s++);
}


char uart1_rx_char(void)
{
    while(!(USART1_SR & (1<<5))); // RXNE
    return USART1_DR;
}



void wait_for_start(void)
{
  char target[] = "START";
  int index = 0;
while(1)
   {
    char c = uart1_rx_char();
    if(c == target[index])
       {
        index++;
        if(index == 5)
        {
        	USART1_CR1 |= (1<<13);
            return;     // START matched
        }
          }
   else
    {
     index = 0;
     }
   }
}


// ========================
// MAIN
// ========================
int main(void){

	RCC_AHB1ENR |= (1U<<0) | (1U<<1) | (1U<<2);
	    RCC_APB2ENR |= (1<<4);


	    GPIOA_MODER &= ~(3<<(9*2));
	    GPIOA_MODER |=  (2<<(9*2));
	    GPIOA_AFRH &= ~(0xF<<(4*(9-8)));
	    GPIOA_AFRH |=  (7<<(4*(9-8)));
	    GPIOA_MODER &= ~(3<<(10*2));
	    GPIOA_MODER |=  (2<<(10*2));
	    GPIOA_AFRH &= ~(0xF<<(4*(10-8)));
	    GPIOA_AFRH |=  (7<<(4*(10-8)));
	    USART1_BRR = 0x0683;
	    USART1_CR1 |= (1<<3);
	    USART1_CR1 |= (1<<2);
	    USART1_CR1 |= (1<<13);

	    GPIOB_MODER &= ~(3U<<(0*2));
	    GPIOB_MODER &= ~(3U<<(1*2));
	    GPIOB_MODER |= (1U<<(1*2));

	    for(int p=9;p<=10;p++)
	    {
	      GPIOB_MODER &= ~(3U<<(p*2));
	      GPIOB_MODER |= (1U<<(p*2));
	    }

	    for(int p=12;p<=15;p++)
	    {
	      GPIOB_MODER &= ~(3U<<(p*2));
	      GPIOB_MODER |= (1U<<(p*2));
	    }

	    for(int p=4;p<=6;p++)
	    {
	      GPIOB_MODER &= ~(3U<<(p*2));
	      GPIOB_MODER |= (1U<<(p*2));
	    }

	    GPIOB_ODR |= (MOTOR_RICE | MOTOR_DHALL | MOTOR_WHEAT);

	    for(int i=1;i<=4;i++)
	    {
	      GPIOC_MODER &= ~(3U<<(i*2));
	      GPIOC_MODER |= (1U<<(i*2));
	    }

	    for(int i=5;i<=7;i++)
	    {
	        GPIOC_MODER &= ~(3U<<(i*2));
	        GPIOC_PUPDR &= ~(3U<<(i*2));
	        GPIOC_PUPDR |= (1U<<(i*2));
	    }

	    for(int i=0;i<=4;i++)
	        {
	    	if(i==3) continue;
	        GPIOA_MODER &= ~(3U<<(i*2));
	        GPIOA_PUPDR &= ~(3U<<(i*2));
	        GPIOA_PUPDR |= (1U<<(i*2));
	        }


    lcd_init();
    lcd_cmd(0x01); lcd_print("VENDING MACHINE");
    delay(8000000);
    lcd_cmd(0x01); lcd_print("Initializing...");
    tare();
    delay(8000000);
    lcd_cmd(0x01); lcd_print("Show your ID");
    delay(8000000);

    while(1)
        {
    	   char selected_item;
    	   char selected_grams;

    	   char *item_name;
    	   char *gram_value;

    	  wait_for_start();
    	  delay(100000);
    	  uart1_tx_char('k');


 // -------- Item Select --------
   lcd_cmd(0x01); lcd_print("Select Items:");
   lcd_cmd(0xC0);
   selected_item = lcd_scroll_left_with_key("1.Rice  2.Dall  3.Sugar       ");


 // -------- IR sensor check --------
   if(selected_item=='1' && (GPIOA_IDR & IR_RICE))
     {
      lcd_cmd(0x01); lcd_print("       RICE");
      lcd_cmd(0xC0); lcd_print(" Out of stock");
      while(keypad_getkey() != '*');
      continue;
     }

   if(selected_item=='2' && (GPIOA_IDR & IR_DALL))
     {
      lcd_cmd(0x01); lcd_print("       DALL");
      lcd_cmd(0xC0); lcd_print(" Out of stock");
      while(keypad_getkey() != '*');
      continue;
     }

   if(selected_item=='3' && (GPIOA_IDR & IR_SUGAR))
     {
      lcd_cmd(0x01); lcd_print("       SUGAR");
      lcd_cmd(0xC0); lcd_print(" Out of stock");
      while(keypad_getkey() != '*');
      continue;
     }

 // -------- Grams Select --------
      lcd_cmd(0x01); lcd_print("Select Grams:");
      lcd_cmd(0xC0);
      selected_grams = lcd_scroll_left_with_key("1.100  2.200  3.300       ");





 /* -------- Decide Item -------- */
                lcd_cmd(0x01);
                if(selected_item=='1')
                {
                    item_name = "RICE";
                    lcd_print("RICE");
                    motor_control(1);
                }
                else if(selected_item=='2')
                {
                    item_name = "DALL";
                    lcd_print("DALL");
                    motor_control(2);
                }
                else
                {
                    item_name = "SUGAR";
                    lcd_print("SUGAR");
                    motor_control(3);
                }

                /* -------- Decide Grams -------- */
                if(selected_grams=='1')
                {
                    gram_value = "100";
                    lcd_print("    100g");
                }
                else if(selected_grams=='2')
                {
                    gram_value = "200";
                    lcd_print("    200g");
                }
                else
                    gram_value = "300";
                    lcd_print("    300g");

                 //-------- Weight Loop --------
                        int targetWeight=0;
                        if(selected_grams=='1') targetWeight=100;
                        else if(selected_grams=='2') targetWeight=200;
                        else if(selected_grams=='3') targetWeight=300;

                        while(1){
                            long sum=0; for(int i=0;i<10;i++) sum+=HX711_Read();
                            long avg=sum/10;
                            int grams = get_weight(avg)*2;

                            lcd_cmd(0xC0);
                            char buf[10];
                            lcd_print("grams: ");
                            sprintf(buf,"%03d",grams);
                            lcd_print(buf);

                            if(grams>=targetWeight)
                            {

                                uart1_tx_string(item_name);
                                uart1_tx_char(',');
                                uart1_tx_string(gram_value);
                                wait_for_continue(grams);
                                break;
                            }
                            else {
                                motor_control(selected_item-'0');
                            }
                            delay(80000);
                        }

                /* -------- Display Final -------- */
                lcd_cmd(0x01);
                lcd_print(item_name);
                lcd_print("     ");
                lcd_print(gram_value);
                lcd_print("g");

                /* -------- UART Send -------- */
                uart1_tx_string(item_name);
                uart1_tx_char(',');
                uart1_tx_string(gram_value);





        delay(8000000);
        lcd_cmd(0x01);
        lcd_print("Show your ID");

    }
