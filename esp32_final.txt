/* -------- Helper Function -------- */
String cleanString(String data)
{
  data.trim();
  String clean = "";

  for (int i = 0; i < data.length(); i++)
  {
    if (isAlphaNumeric(data[i]) || data[i] == ',')
      clean += data[i];
  }

  return clean;
}

void saveHistoryToEEPROM() {
  EEPROM.put(ADDR_HISTORY_HEAD, historyHead);
  EEPROM.put(ADDR_HISTORY_COUNT, historyCount);

  for (int i = 0; i < MAX_HISTORY; i++) {
    int addr = ADDR_HISTORY_BASE + (i * HISTORY_RECORD_SIZE);
    EEPROM.put(addr, history[i]);
  }

  EEPROM.commit();
  Serial.println("💾 History saved to EEPROM");
}

void loadHistoryFromEEPROM() {
  EEPROM.get(ADDR_HISTORY_HEAD, historyHead);
  EEPROM.get(ADDR_HISTORY_COUNT, historyCount);

  if (historyHead < 0 || historyHead >= MAX_HISTORY) historyHead = 0;
  if (historyCount < 0 || historyCount > MAX_HISTORY) historyCount = 0;

  for (int i = 0; i < MAX_HISTORY; i++) {
    int addr = ADDR_HISTORY_BASE + (i * HISTORY_RECORD_SIZE);
    EEPROM.get(addr, history[i]);

    // safety cleanup
    if (history[i].qty < 0 || history[i].qty > 50000) {
      history[i].qty = 0;
      strcpy(history[i].user, "");
      strcpy(history[i].item, "");
    }
  }
  Serial.print("History Count: ");
  Serial.println(historyCount);
  Serial.println("✅ History loaded safely");
}

void clearHistoryEEPROM() {
  historyHead = 0;
  historyCount = 0;

  for (int i = 0; i < MAX_HISTORY; i++) {
    history[i].qty = 0;
    strcpy(history[i].user, "");
    strcpy(history[i].item, "");

    int addr = ADDR_HISTORY_BASE + (i * HISTORY_RECORD_SIZE);
    EEPROM.put(addr, history[i]);
  }

  EEPROM.put(ADDR_HISTORY_HEAD, historyHead);
  EEPROM.put(ADDR_HISTORY_COUNT, historyCount);
  EEPROM.commit();

  Serial.println("🧹 History EEPROM cleared");
}

void waitForNetwork() {
  gsmReady = false;

  Serial.println("⏳ Waiting for GSM network...");
  unsigned long start = millis();

  while (millis() - start < 40000) {
    gsm.println("AT+CREG?");
    delay(1500);

    String resp = "";
    while (gsm.available()) resp += char(gsm.read());

    Serial.println(resp);

    if (resp.indexOf("+CREG: 1") != -1 || resp.indexOf("+CREG: 5") != -1) {
      Serial.println("✅ GSM Network Registered!");
      gsmReady = true;
      return;
    }
  }

  Serial.println("❌ GSM Network not found (timeout)");
}

void readGSM(unsigned long timeout = 3000) {
  unsigned long t = millis();
  while (millis() - t < timeout) {
    while (gsm.available()) {
      Serial.write(gsm.read());
    }
  }
}

void sendSMS(const char* number, String message)
{
  // Set SMS text mode
  gsm.println("AT+CMGF=1");
  delay(1000);

  // Send number
  gsm.print("AT+CMGS=\"");
  gsm.print(number);
  gsm.println("\"");

  // 🔹 WAIT FOR '>' PROMPT
  bool prompt = false;
  unsigned long t = millis();

  while (millis() - t < 5000) {
    if (gsm.available()) {
      char c = gsm.read();
      if (c == '>') {
        prompt = true;
        break;
      }
    }
  }

  if (!prompt) {
    Serial.println("❌ GSM prompt not received");
    return;
  }

  // 🔹 Send message ONLY ONCE
  gsm.print(message);
  delay(500);

  gsm.write(26);   // CTRL+Z to send
  Serial.println("📤 Sending SMS...");

  // Wait for response
  readGSM(8000);

  Serial.println("✅ SMS Process Completed");
}

void loop() {
  server.handleClient();

  if (!mfrc522.PICC_IsNewCardPresent()) return;
  if (!mfrc522.PICC_ReadCardSerial()) return;

  uidString = "";
  for (byte i = 0; i < mfrc522.uid.size; i++) 
  {
    if (mfrc522.uid.uidByte[i] < 0x10) uidString += "0";
    uidString += String(mfrc522.uid.uidByte[i], HEX);
    uidString += " ";
  }
  uidString.toUpperCase();

  if (uidString == "23 37 7B 38 ") customerName = "KEERTHANA";
  else if (uidString == "44 DF 51 2E ") customerName = "KARTHIK";
  else if (uidString == "23 8F 34 15 ") customerName = "KISHORE";
  else customerName = "UNKNOWN";

  Serial.println("\n==============================");
  Serial.println("🪪 RFID Detected: " + uidString);
  Serial.println("👤 Customer: " + customerName);
  if (customerName == "UNKNOWN") {
  Serial.println("❌ Unknown Card - Access Denied");
  Serial.println("==============================\n");
  mfrc522.PICC_HaltA();
  delay(3000);
  return;
}

    // 🔒 Reject if balance already negative (DO NOT start weighing)
  float currentBalance = 0;

  if (customerName == "KEERTHANA") currentBalance = balanceKEERTHANA;
  else if (customerName == "KISHORE") currentBalance = balanceKISHORE;
  else if (customerName == "KARTHIK") currentBalance = balanceKARTHIK;

  if (currentBalance < 0) {
    Serial.println("❌ Purchase Blocked: Negative Balance");
    Serial.println("💳 Current Balance: ₹" + String(currentBalance, 2));
    Serial.println("🚫 Weighing module NOT started");
    Serial.println("==============================\n");

    mfrc522.PICC_HaltA();   // stop RFID
    delay(3000);
    return;                // exit loop → START will NOT be sent
  }

/* -------- Clear UART Buffer -------- */
while (Serial1.available())
  Serial1.read();

delay(100);

/* -------- Send START -------- */
Serial1.println("START");
Serial.println("START sent");
Serial.println("Waiting for ACK...");

/* -------- Wait for ACK -------- */
bool ackReceived = false;
unsigned long ackStart = millis();

while (millis() - ackStart < 5000)
{
  if (Serial1.available())
  {
    String ackLine = Serial1.readStringUntil('\n');
    ackLine.trim();

    Serial.println("UART ACK Raw: " + ackLine);

    if (ackLine == "k")
    {
      ackReceived = true;
      break;
    }
  }
}

if (!ackReceived)
{
  Serial.println("❌ ACK not received. Cancelling transaction.");
  mfrc522.PICC_HaltA();
  delay(2000);
  return;
}

Serial.println("✅ ACK received. Waiting for item data...");


/* -------- Wait for ITEM,QUANTITY -------- */
bool validData = false;
unsigned long dataStart = millis();
rxData = "";

while (millis() - dataStart < 100000)
{
  if (Serial1.available())
  {
    rxData = Serial1.readStringUntil('\n');
    rxData = cleanString(rxData);

    Serial.println("UART Data Raw: " + rxData);

    if (rxData.indexOf(',') > 0)
    {
      validData = true;
      break;
    }
  }
}

if (!validData)
{
  Serial.println("❌ No valid item data received.");
  mfrc522.PICC_HaltA();
  delay(2000);
  return;
}

/* -------- Parse -------- */
int c = rxData.indexOf(',');
item = rxData.substring(0, c);
int qty = rxData.substring(c + 1).toInt();

item.trim();
item.toUpperCase();

/* -------- Validate Item -------- */
if (item != "RICE" && item != "DALL" && item != "SUGAR")
{
  Serial.println("❌ Invalid item received!");
  mfrc522.PICC_HaltA();
  delay(2000);
  return;
}

  float pricePerGram = (item == "RICE") ? ricePrice : (item == "DALL") ? dallPrice : sugarPrice;
  float totalPrice = qty * pricePerGram;

  Serial.println("🛒 Item: " + item);
  Serial.println("⚖ Quantity: " + String(qty) + " g");
